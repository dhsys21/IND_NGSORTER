▣ 제품 사양 
• 사용 전압 : DC 24V 
• 소비 전류 :  100mA 
• 사용 온도 : -10℃ ~ 50℃
• 크기 : 106mm(W) * 78mm(D) * 39mm(H) 
• 케이스 재질 : ABS 난연 
• 알람 경고등 : 적색(연기 알람), 황색(온도 알람) / Two Color LED 
• 연기 감지 방식 : 광전식 
• 온도 검출 방식 : NTC-10KD-5 써미스터(온도 센서) 
• 출력 형태 : NPN Open Collector & RS485 
• 프로토콜 : HumanAutomation Protocol 
• 프로토콜 : Modbus RTU (Remote Terminal Unit) Protocol 
• 출력 최대 전류 : DC30V   MAX. 50mA 
• 외부 인터페이스 : PC, 산업용 제어기, PLC 등 DIGITAL I/O & SERIAL 통신 
• 규격 승인 : CE 승인 (승인 검사서 요청 시 별도 발송)
• 신뢰성 시험 : 환경 신뢰성 검사 ( 신뢰성 검사서 요청 시 별도 발송)

//////////////////////////////////////////////////////////////////////////////////////// 
unsigned short get_crc16( unsigned char* pbuf, int len )
{
unsigned short crc=0xFFFF; 
while (len--)
{
crc ^= *pbuf++; 
for (int i=0; i<8; i++) 
{
if (crc & 0x01) 
{
crc >>= 1; 
crc ^= 0xA001; 
}
else 
{
crc >>= 1; 
}
}
}
return crc;
}
//////////////////////////////////////////////////////////////////////////////////////// 
void main(void) 
{
unsigned char pTxBuf[] = { 0x01, 0x03, 0x10, 0x00, 0x00, 0x04, 0, 0 }; 
unsigned short crc16; 
crc16 = get_crc16(pTxBuf, 6); 
pTxBuf[6] = (unsigned char)((crc16) & 0x00FF); 
pTxBuf[7] = (unsigned char)((crc16 >> 8) & 0x00FF);
